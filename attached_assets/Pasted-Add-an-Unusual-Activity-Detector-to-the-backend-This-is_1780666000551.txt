CHANGE 1 — add computeUnusualActivity()
═══════════════════════════════════════════
Add this function near the signal scanner. It scores EACH tracked symbol 0–100 by blending normalized sub-signals. Each sub-signal is 0..1, weighted, summed, clamped. NEVER frame output as a prediction.

    // ── UNUSUAL ACTIVITY DETECTOR ────────────────────────────────────
    // Probabilistic "conditions" score. NOT a forecast. Flags abnormal
    // positioning/activity that historically PRECEDES volatility — with
    // high false-positive rate. Education/information only.
    type UnusualSignal = {
      symbol: string;
      score: number;            // 0..100
      band: "QUIET" | "ELEVATED" | "HIGH" | "EXTREME";
      reasons: string[];        // which sub-signals fired, plain English
      components: Record<string, number>; // 0..1 each, for transparency
      ts: number;
    };

    function computeUnusualActivity(): UnusualSignal[] {
      const now = Date.now();
      const out: UnusualSignal[] = [];
      const clamp01 = (x: number) => Math.max(0, Math.min(1, x));

      for (const sym of Object.keys(priceHistory)) {
        const hist = priceHistory[sym];
        if (!hist || hist.length < 8) continue;

        const current = hist[hist.length - 1];

        // --- 1) ACCELERATION: short-window return vs medium-window baseline
        const shortPts = hist.filter(p => now - p.ts <= 120000);   // ~2min
        const medPts   = hist.filter(p => now - p.ts <= 600000);   // ~10min
        const shortStart = shortPts.length ? shortPts[0] : current;
        const medStart   = medPts.length ? medPts[0] : current;
        const shortRet = shortStart.price ? Math.abs((current.price - shortStart.price) / shortStart.price) : 0;
        const medRet   = medStart.price ? Math.abs((current.price - medStart.price) / medStart.price) : 0;
        // accel = how much the last 2min outpaces the 10min run-rate (per-min)
        const shortPerMin = shortRet / 2;
        const medPerMin   = medRet / 10 || 0.0001;
        const accel = clamp01((shortPerMin / medPerMin - 1) / 3); // 4x run-rate => ~1.0

        // --- 2) TICK ACTIVITY SURGE (proxy — NOT true traded volume)
        const recent = hist.filter(p => now - p.ts <= 180000).length; // last 3min ticks
        const older  = hist.filter(p => now - p.ts > 180000 && now - p.ts <= 900000).length; // prior 12min
        const olderRate = older > 0 ? older / 4 : 1; // normalize to 3min-equivalent
        const activityMult = olderRate > 0 ? recent / olderRate : 1;
        const activity = clamp01((activityMult - 1.5) / 2.5); // 1.5x..4x => 0..1

        // --- 3) FUNDING EXTREMITY (crowded positioning => squeeze fuel)
        // Read funding for this symbol from the same perp source /api/signals uses.
        // Replace `getFundingFor(sym)` with the actual lookup in this file
        // (e.g. perpMap[base]?.funding). Funding is per-8h decimal (e.g. 0.0003).
        const funding = (function () {
          try { return getFundingFor(sym); } catch { return 0; }
        })();
        const fundingExtremity = clamp01((Math.abs(funding) - 0.0005) / 0.0025); // |0.05%|..|0.30%|

        // --- 4) OPEN INTEREST PRESENCE (more OI => more fuel if it moves)
        const oi = (function () {
          try { return getOIFor(sym); } catch { return 0; }
        })();
        const oiWeight = clamp01((oi - 5e6) / 95e6); // $5M..$100M => 0..1

        // --- 5) RANGE COMPRESSION→EXPANSION (optional, via ATR if available)
        let expansion = 0;
        try {
          const atr = calcATR(sym);
          if (atr > 0 && current.price > 0) {
            const atrPct = atr / current.price;
            expansion = clamp01((atrPct - 0.004) / 0.02); // 0.4%..2.4% ATR
          }
        } catch {}

        const components = {
          acceleration: accel,
          tickActivity: activity,
          fundingExtremity,
          openInterest: oiWeight,
          rangeExpansion: expansion,
        };

        // Weights — acceleration + activity dominate; funding/OI are context.
        const score100 = Math.round(
          (accel * 0.34 +
           activity * 0.26 +
           fundingExtremity * 0.20 +
           oiWeight * 0.10 +
           expansion * 0.10) * 100
        );

        if (score100 < 35) continue; // only surface meaningful readings

        const reasons: string[] = [];
        if (accel >= 0.4) reasons.push(`Price velocity accelerating (${(shortPerMin*100).toFixed(2)}%/min vs ${(medPerMin*100).toFixed(2)}%/min baseline)`);
        if (activity >= 0.4) reasons.push(`Tick activity ${activityMult.toFixed(1)}x recent baseline (proxy, not traded volume)`);
        if (fundingExtremity >= 0.4) reasons.push(`Funding extreme at ${(funding*100).toFixed(4)}%/8h — crowded ${funding >= 0 ? "longs" : "shorts"}, squeeze risk`);
        if (oiWeight >= 0.5) reasons.push(`Open interest $${(oi/1e6).toFixed(0)}M — elevated positioning`);
        if (expansion >= 0.4) reasons.push(`Volatility range expanding`);
        if (reasons.length === 0) reasons.push("Mild composite anomaly across multiple inputs");

        const band: UnusualSignal["band"] =
          score100 >= 80 ? "EXTREME" :
          score100 >= 65 ? "HIGH" :
          score100 >= 50 ? "ELEVATED" : "QUIET";

        out.push({ symbol: sym, score: score100, band, reasons, components, ts: now });
      }

      out.sort((a, b) => b.score - a.score);
      return out.slice(0, 20);
    }

IMPORTANT for Replit: the two placeholders getFundingFor(sym) and getOIFor(sym) must be replaced with the ACTUAL in-file lookups the scanner already uses for funding/OI (e.g. perpMap[base]?.funding and perpMap[base]?.oi, mapping sym→base ticker the same way the scanner does). If a clean lookup isn't in scope, wire the perp map into this function. Do not fetch new data.

CHANGE 2 — add the endpoint
═══════════════════════════════════════════
In registerRoutes, alongside /api/signals and /api/regime, add a cached endpoint (30s cache like /api/regime):

    let unusualCache: { data: any; ts: number } | null = null;
    app.get("/api/unusual", (_req, res) => {
      if (unusualCache && Date.now() - unusualCache.ts < 30000) {
        return res.json(unusualCache.data);
      }
      const signals = computeUnusualActivity();
      const data = {
        signals,
        count: signals.length,
        ts: Date.now(),
        disclaimer: "Unusual Activity flags abnormal market conditions that can precede volatility. This is information and education only — NOT a prediction, signal to trade, or financial advice. High false-positive rate. Always do your own research.",
      };
      unusualCache = { data, ts: Date.now() };
      res.json(data);
    });

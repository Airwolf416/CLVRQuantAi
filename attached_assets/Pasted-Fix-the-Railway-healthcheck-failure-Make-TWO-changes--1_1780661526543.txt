If any of those throw or hang (e.g. Postgres in recovery mode), the port never binds and Railway's healthcheck fails. (b) Find the bottom block that looks like:

      ... error middleware ...
      ... serveStatic / setupVite ...
      const port = parseInt(process.env.PORT || "5000", 10);

Replace that ENTIRE block with:

    const port = parseInt(process.env.PORT || "5000", 10);

    httpServer.listen({ port, host: "0.0.0.0", reusePort: true }, () => {
      log(`serving on port ${port}`);
    });

    process.on("unhandledRejection", (reason) => {
      console.error("UNHANDLED REJECTION:", reason);
    });
    process.on("uncaughtException", (err) => {
      console.error("UNCAUGHT EXCEPTION:", err);
    });

    (async () => {
      try {
        await initializeDatabase();
        await initStripe();
        await registerRoutes(httpServer, app);
        startDailyBriefScheduler();

        app.use((err: any, _req: Request, res: Response, next: NextFunction) => {
          const status = err.status || err.statusCode || 500;
          const message = err.message || "Internal Server Error";
          console.error("Internal Server Error:", err);
          if (res.headersSent) return next(err);
          return res.status(status).json({ message });
        });

        if (process.env.NODE_ENV === "production") {
          serveStatic(app);
        } else {
          const { setupVite } = await import("./vite");
          await setupVite(httpServer, app);
        }

        log("startup complete");
      } catch (err) {
        console.error("STARTUP ERROR:", err);
        // Server stays up on /health so Railway stops crash-looping and logs stay readable.
      }
    })();

PRICE RESOLVER (authoritative)
═══════════════════════════════════════════
    async function resolveSessionPrice(userId: string) {
      const u = await storage.getUser(userId);
      const tier = u?.tier || "free";
      if (tier === "elite") {
        const used = u?.eliteFreeSessionsUsed ?? 0;
        if (used < ELITE_FREE_LIFETIME)
          return { cents: 0, isFreeElite: true, tier, freeRemaining: ELITE_FREE_LIFETIME - used };
        return { cents: SESSION_PRO_CENTS, isFreeElite: false, tier, freeRemaining: 0 };
      }
      if (tier === "pro") return { cents: SESSION_PRO_CENTS, isFreeElite: false, tier, freeRemaining: null };
      return { cents: SESSION_FREE_CENTS, isFreeElite: false, tier, freeRemaining: null };
    }

ENDPOINTS (all require auth; reject with 401 if no session userId)
═══════════════════════════════════════════
GET /api/concierge/pricing
  → { tier, priceCents, priceDisplay: `$${(priceCents/100).toFixed(0)}`, isFreeElite, freeRemaining }

POST /api/concierge/book   body: { slotStart: ISO, slotEnd: ISO }
  - Resolve price server-side via resolveSessionPrice(userId). IGNORE any price/tier in the body.
  - Validate slotStart is in the future and slotEnd > slotStart; reject otherwise.
  - If isFreeElite (cents === 0):
      * insert booking (status 'confirmed', priceCents 0, isFreeElite true, tier, slots)
      * increment users.eliteFreeSessionsUsed by 1
      * return { mode: "free_confirmed", bookingId }
  - Else (paid):
      * insert booking (status 'pending', priceCents = cents, tier, slots)
      * create a Stripe Checkout Session in PAYMENT mode (one-off, NOT subscription):
          line_items: [{ price_data: { currency:'usd', unit_amount: cents,
            product_data: {
              name: '30-Min CLVRQuant Platform Training',
              description: 'Live 1-on-1 walkthrough of the CLVRQuant platform. Educational/training session — not personalized financial advice.'
            } }, quantity: 1 }]
          mode: 'payment',
          success_url / cancel_url back into the app,
          metadata: { bookingId, userId, type: 'session_booking' }
      * save stripeSessionId on the booking row
      * return { mode: "checkout", url: session.url }

GET /api/concierge/bookings  → current user's bookings (most recent first), for "my sessions" display.

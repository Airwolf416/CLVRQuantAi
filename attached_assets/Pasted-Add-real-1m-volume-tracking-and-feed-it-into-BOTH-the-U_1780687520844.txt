1) If not already present (search "volHistory"/"getVolumeZ"/"pollVolumes"), add near the scanner scope:

    const volHistory: Record<string, number[]> = {};
    async function pollVolumes() {
      for (const [base, symbol] of Object.entries(BINANCE_SYMBOLS)) {
        try {
          const r = await fetch(`https://api.binance.com/api/v3/klines?symbol=${symbol}&interval=1m&limit=31`, { signal: AbortSignal.timeout(5000) });
          const data: any = await r.json();
          if (!Array.isArray(data) || data.length < 6) continue;
          const vols = data.slice(0, -1).map((c: any) => parseFloat(c[7])).filter((v: number) => v > 0);
          if (vols.length >= 5) volHistory[base] = vols.slice(-30);
        } catch {}
        await delay(150);
      }
    }
    function getVolumeZ(sym: string): { z: number; ratio: number } {
      const v = volHistory[sym];
      if (!v || v.length < 8) return { z: 0, ratio: 0 };
      const last = v[v.length - 1];
      const prior = v.slice(0, -1);
      const mean = prior.reduce((a, b) => a + b, 0) / prior.length;
      const sd = Math.sqrt(prior.reduce((a, b) => a + (b - mean) ** 2, 0) / prior.length) || 1;
      return { z: (last - mean) / sd, ratio: mean > 0 ? last / mean : 0 };
    }

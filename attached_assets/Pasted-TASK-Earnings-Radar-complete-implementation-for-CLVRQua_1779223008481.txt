async function run() {
  const client = await pool.connect();
  try {
    await client.query(`
      CREATE TABLE IF NOT EXISTS schema_migrations (
        filename TEXT PRIMARY KEY,
        applied_at TIMESTAMPTZ DEFAULT NOW()
      )
    `);
    const files = fs.readdirSync(__dirname).filter(f => f.endsWith('.sql')).sort();
    for (const f of files) {
      const { rows } = await client.query(
        'SELECT 1 FROM schema_migrations WHERE filename=$1', [f]);
      if (rows.length) { console.log(`✓ ${f} (already applied)`); continue; }
      const sql = fs.readFileSync(path.join(__dirname, f), 'utf8');
      console.log(`→ applying ${f}...`);
      await client.query(sql);
      await client.query('INSERT INTO schema_migrations (filename) VALUES ($1)', [f]);
      console.log(`✓ ${f} applied`);
    }
    console.log('All migrations applied.');
  } catch (e) { console.error('Migration failed:', e); process.exit(1); }
  finally { client.release(); await pool.end(); }
}

import os, json, asyncio, logging

LOOKAHEAD_DAYS = 5


async def fetch_earnings_calendar() -> list[dict]:
    start = datetime.utcnow().date()
    end = start + timedelta(days=LOOKAHEAD_DAYS)
    url = (f"https://financialmodelingprep.com/api/v3/earning_calendar"
           f"?from={start}&to={end}&apikey={FMP_KEY}")
    async with httpx.AsyncClient(timeout=20) as c:
        r = await c.get(url); r.raise_for_status()
        return r.json()


async def compute_features(symbol: str, client: httpx.AsyncClient) -> Optional[dict]:
    try:
        surprises = (await client.get(
            f"https://financialmodelingprep.com/api/v3/earnings-surprises/"
            f"{symbol}?apikey={FMP_KEY}")).json()[:8]
        prices = (await client.get(
            f"https://financialmodelingprep.com/api/v3/historical-price-full/"
            f"{symbol}?serietype=line&apikey={FMP_KEY}")).json().get("historical", [])
        if not surprises or len(prices) < 30: return None

        moves, beats, drift_beat, drift_miss = [], 0, [], []
        for s in surprises:
            edate = s["date"]
            actual = s.get("actualEarningResult"); est = s.get("estimatedEarning")
            if actual is None or est is None: continue
            idx = next((i for i,p in enumerate(prices) if p["date"] <= edate), None)
            if idx is None or idx + 5 >= len(prices): continue
            p_t = prices[idx]["close"]
            p_t1 = prices[max(idx-1,0)]["close"]
            p_t5 = prices[max(idx-5,0)]["close"]
            moves.append(abs(p_t1/p_t - 1) * 100)
            if actual > est:
                beats += 1; drift_beat.append((p_t5/p_t - 1) * 100)
            else:
                drift_miss.append((p_t5/p_t - 1) * 100)

        hist_avg_move = sum(moves)/len(moves) if moves else 0
        beat_rate = beats/len(surprises) * 100 if surprises else 0

        implied_move = await _implied_move(symbol, client)

        rev = (await client.get(
            f"https://financialmodelingprep.com/api/v4/upgrades-downgrades-rss-feed"
            f"?symbol={symbol}&apikey={FMP_KEY}")).json()
        revisions = sum(1 if r.get("newGrade","").lower() in ("buy","outperform") else -1
                        for r in rev[:20])

        profile = (await client.get(
            f"https://financialmodelingprep.com/api/v3/profile/"
            f"{symbol}?apikey={FMP_KEY}")).json()
        sector_etf = _sector_to_etf(profile[0].get("sector") if profile else "")
        sector_rs = await _relative_strength(symbol, sector_etf, client)

        return {
            "hist_avg_move_1d": round(hist_avg_move, 2),
            "beat_rate_8q": round(beat_rate, 1),
            "drift_beat_5d": round(sum(drift_beat)/len(drift_beat), 2) if drift_beat else 0,
            "drift_miss_5d": round(sum(drift_miss)/len(drift_miss), 2) if drift_miss else 0,
            "implied_move_pct": implied_move,
            "revisions_90d": revisions,
            "sector_rs_30d": sector_rs,
            "macro_overlay": await _macro_events_this_week(client),
            "liquidity_score": _liquidity(prices, profile),
        }
    except Exception as e:
        log.exception(f"feature compute failed {symbol}: {e}")
        return None


async def _implied_move(symbol, client):
    try:
        opts = (await client.get(
            f"https://financialmodelingprep.com/api/v4/options/"
            f"{symbol}?apikey={FMP_KEY}")).json()
        if not opts: return 0
        atm = min(opts, key=lambda o: abs(o.get("strike",0) - o.get("underlying",0)))
        straddle = atm.get("call_price",0) + atm.get("put_price",0)
        spot = atm.get("underlying", 1) or 1
        return round(straddle / spot * 100, 2)
    except Exception: return 0


def _sector_to_etf(sector: str) -> str:
    return {
        "Technology":"XLK","Financial Services":"XLF","Healthcare":"XLV",
        "Consumer Cyclical":"XLY","Consumer Defensive":"XLP","Energy":"XLE",
        "Industrials":"XLI","Communication Services":"XLC","Utilities":"XLU",
        "Real Estate":"XLRE","Basic Materials":"XLB",
    }.get(sector, "SPY")


async def _relative_strength(symbol, etf, client):
    async def ret30(s):
        p = (await client.get(
            f"https://financialmodelingprep.com/api/v3/historical-price-full/"
            f"{s}?serietype=line&apikey={FMP_KEY}")).json().get("historical", [])
        if len(p) < 30: return 0
        return (p[0]["close"] / p[29]["close"] - 1) * 100
    a, b = await asyncio.gather(ret30(symbol), ret30(etf))
    return round(a - b, 2)


async def _macro_events_this_week(client):
    today = datetime.utcnow().date(); end = today + timedelta(days=7)
    try:
        evs = (await client.get(
            f"https://financialmodelingprep.com/api/v3/economic_calendar"
            f"?from={today}&to={end}&apikey={FMP_KEY}")).json()
        return [e["event"] for e in evs
                if e.get("country")=="US" and e.get("impact")=="High"][:5]
    except Exception: return []


def _liquidity(prices, profile):
    if not prices or not profile: return 0
    avg_vol = sum(p.get("volume",0) for p in prices[:20]) / min(20, len(prices))
    avg_px  = sum(p["close"] for p in prices[:20]) / min(20, len(prices))
    mcap = profile[0].get("mktCap", 1) or 1
    return round((avg_vol * avg_px) / mcap, 4)

"""Polls FMP for actual results during reporting windows.
Publishes to Redis: earnings:results, squawk:alerts, push:earnings."""
import os, json, asyncio, logging
from datetime import datetime, timezone, timedelta
import asyncpg, httpx
import redis.asyncio as redis
from earnings_analyzer import get_post_earnings_analysis

log = logging.getLogger("earnings_poller")
FMP_KEY = os.environ["FMP_API_KEY"]
DB_URL = os.environ["DATABASE_URL"]
REDIS_URL = os.environ["REDIS_URL"]

POLL_INTERVAL_S = 30


def in_reporting_window():
    now = datetime.now(timezone(timedelta(hours=-5)))  # ET (adjust DST as needed)
    cur = now.hour * 60 + now.minute
    return (6*60 <= cur <= 9*60+30) or (16*60 <= cur <= 20*60)


async def get_pending_symbols(pool):
    today = datetime.utcnow().date()
    rows = await pool.fetch("""
        SELECT symbol, report_time, eps_estimate, revenue_estimate,
               ai_analysis, company_name
        FROM earnings_cache
        WHERE report_date = $1
          AND (ai_analysis->>'actual_posted')::boolean IS NOT TRUE
    """, today)
    return [dict(r) for r in rows]


async def fetch_actual(symbol, client):
    try:
        r = await client.get(
            f"https://financialmodelingprep.com/api/v3/earnings-surprises/"
            f"{symbol}?apikey={FMP_KEY}", timeout=10)
        data = r.json()
        if not data: return None
        latest = data[0]
        rep_date = datetime.fromisoformat(latest["date"]).date()
        if rep_date != datetime.utcnow().date(): return None
        if latest.get("actualEarningResult") is None: return None
        return {"actual_eps":latest["actualEarningResult"],
                "eps_estimate":latest.get("estimatedEarning"),
                "report_date":latest["date"]}
    except Exception as e:
        log.warning(f"fetch_actual {symbol}: {e}"); return None


async def fetch_revenue_and_price(symbol, client):
    out = {"actual_revenue":None,"price_reaction_pct":0}
    try:
        inc = (await client.get(
            f"https://financialmodelingprep.com/api/v3/income-statement/"
            f"{symbol}?period=quarter&limit=1&apikey={FMP_KEY}")).json()
        if inc: out["actual_revenue"] = inc[0].get("revenue")
        q = (await client.get(
            f"https://financialmodelingprep.com/api/v3/quote/"
            f"{symbol}?apikey={FMP_KEY}")).json()
        if q: out["price_reaction_pct"] = round(q[0].get("changesPercentage",0), 2)
    except Exception: pass
    return out


async def process_result(symbol, meta, actual, pool, rds, client):
    extra = await fetch_revenue_and_price(symbol, client)
    eps_surprise = 0
    if meta.get("eps_estimate") and actual["actual_eps"]:
        est = float(meta["eps_estimate"])
        if est != 0:
            eps_surprise = round((actual["actual_eps"] - est) / abs(est) * 100, 2)

    payload = {**actual, **extra,
               "eps_surprise_pct":eps_surprise,
               "revenue_estimate":meta.get("revenue_estimate"),
               "released_at":datetime.utcnow().isoformat()}

    pre_ai = meta["ai_analysis"] if isinstance(meta["ai_analysis"], dict) \
             else json.loads(meta["ai_analysis"])
    post_ai = await get_post_earnings_analysis(symbol, pre_ai, payload)

    merged = {**pre_ai, "actual_posted":True,
              "actual":payload, "post_earnings":post_ai}
    await pool.execute("""
        UPDATE earnings_cache SET ai_analysis=$1, computed_at=NOW()
        WHERE symbol=$2 AND report_date=$3
    """, json.dumps(merged), symbol, datetime.utcnow().date())

    event = {"type":"earnings_result","symbol":symbol,
             "company_name":meta["company_name"],
             "eps_surprise_pct":eps_surprise,
             "price_reaction_pct":extra["price_reaction_pct"],
             "pre_verdict":pre_ai.get("verdict"),
             "call_outcome":post_ai.get("call_outcome"),
             "next_bias":post_ai.get("next_session_bias"),
             "thesis":post_ai.get("thesis"),
             "timestamp":payload["released_at"]}
    await rds.publish("earnings:results", json.dumps(event))

    if abs(eps_surprise) >= 5:
        await rds.publish("squawk:alerts", json.dumps({
            "priority":"high",
            "text":f"{symbol} {'beat' if eps_surprise>0 else 'missed'} by "
                   f"{abs(eps_surprise):.1f} percent. Stock {extra['price_reaction_pct']:+.1f}%. "
                   f"Our pre-earnings call was {pre_ai.get('verdict')}."}))

    await rds.publish("push:earnings", json.dumps({
        "title":f"{symbol} earnings {'BEAT' if eps_surprise>0 else 'MISS'}",
        "body":f"EPS surprise {eps_surprise:+.1f}% • {post_ai.get('thesis','')[:80]}",
        "deep_link":f"/earnings?symbol={symbol}"}))

    log.info(f"processed {symbol} surprise={eps_surprise}% outcome={post_ai.get('call_outcome')}")


async def poll_loop():
    pool = await asyncpg.create_pool(DB_URL)
    rds = redis.from_url(REDIS_URL)
    async with httpx.AsyncClient() as client:
        while True:
            try:
                if not in_reporting_window():
                    await asyncio.sleep(60); continue
                pending = await get_pending_symbols(pool)
                for meta in pending:
                    actual = await fetch_actual(meta["symbol"], client)
                    if actual:
                        await process_result(meta["symbol"], meta, actual,
                                             pool, rds, client)
                await asyncio.sleep(POLL_INTERVAL_S)
            except Exception as e:
                log.exception(f"poll loop error: {e}")
                await asyncio.sleep(POLL_INTERVAL_S)


if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO)
    asyncio.run(poll_loop())

    ORDER BY market_cap DESC`, [today]);

});

router.get('/week', requireProOrElite, async (req, res) => {
  const start = new Date().toISOString().slice(0,10);
  const end = new Date(Date.now() + 5*86400000).toISOString().slice(0,10);
  const { rows } = await pool.query(`
    SELECT report_date, symbol, company_name, market_cap, report_time,
           eps_estimate, ai_analysis
    FROM earnings_cache WHERE report_date BETWEEN $1 AND $2
    ORDER BY report_date, market_cap DESC`, [start, end]);
  const grouped = rows.reduce((acc, r) => {
    const d = r.report_date.toISOString().slice(0,10);
    (acc[d] = acc[d] || []).push(r); return acc;
  }, {});
  res.json({ grouped });
});

router.get('/analysis/:symbol', requireProOrElite, async (req, res) => {
  const today = new Date().toISOString().slice(0,10);
  const { rows } = await pool.query(`
    SELECT * FROM earnings_cache
    WHERE symbol = $1 AND report_date >= $2
    ORDER BY report_date ASC LIMIT 1`,
    [req.params.symbol.toUpperCase(), today]);
  if (!rows.length) return res.status(404).json({ error:'not_found' });
  res.json(rows[0]);
});

router.get('/teaser', async (req, res) => {
  const today = new Date().toISOString().slice(0,10);
  const { rows } = await pool.query(`
    SELECT COUNT(*)::int AS total,
           COUNT(*) FILTER (WHERE ai_analysis->>'verdict' != 'AVOID')::int AS actionable
    FROM earnings_cache WHERE report_date = $1`, [today]);
  res.json(rows[0]);
});

const WebSocket = require('ws');
const Redis = require('ioredis');

const CHANNELS = ['earnings:results', 'signals:new', 'squawk:alerts'];

function attachWS(server) {
  const wss = new WebSocket.Server({ server, path:'/ws' });
  const sub = new Redis(process.env.REDIS_URL);

  wss.on('connection', (ws) => {
    ws.channels = new Set();
    ws.on('message', (raw) => {
      let msg; try { msg = JSON.parse(raw); } catch { return; }
      if (msg.type === 'subscribe') ws.channels.add(msg.channel);
      if (msg.type === 'unsubscribe') ws.channels.delete(msg.channel);
      if (msg.type === 'ping') ws.send(JSON.stringify({ channel:'pong' }));
    });
    ws.on('close', () => ws.channels.clear());
  });

  sub.subscribe(...CHANNELS, (err) => {
    if (err) console.error('Redis subscribe failed:', err);
    else console.log(`WS bridge subscribed: ${CHANNELS.join(', ')}`);
  });

  sub.on('message', (channel, payload) => {
    let data; try { data = JSON.parse(payload); } catch { data = payload; }
    const frame = JSON.stringify({ channel, data });
    wss.clients.forEach((c) => {
      if (c.readyState === WebSocket.OPEN && c.channels.has(channel)) c.send(frame);
    });
  });
  return wss;
}

import { useEffect, useRef } from 'react';

type Message = { channel: string; data: any };
type Handler = (data: any) => void;

class WSBus {
  private ws: WebSocket | null = null;
  private url: string;
  private handlers = new Map<string, Set<Handler>>();
  private reconnectAttempts = 0;
  private reconnectTimer: any = null;
  private heartbeatTimer: any = null;
  private connected = false;

  constructor(url: string) { this.url = url; this.connect(); }

  private connect() {
    try { this.ws = new WebSocket(this.url); }
    catch { this.scheduleReconnect(); return; }

    this.ws.onopen = () => {
      this.connected = true; this.reconnectAttempts = 0;
      for (const channel of this.handlers.keys())
        this.send({ type:'subscribe', channel });
      this.heartbeatTimer = setInterval(() => {
        if (this.ws?.readyState === WebSocket.OPEN) this.send({ type:'ping' });
      }, 25000);
    };
    this.ws.onmessage = (evt) => {
      let msg: Message; try { msg = JSON.parse(evt.data); } catch { return; }
      if (msg.channel === 'pong') return;
      const set = this.handlers.get(msg.channel);
      if (set) set.forEach(h => { try { h(msg.data); } catch (e) { console.error(e); } });
    };
    this.ws.onclose = () => {
      this.connected = false; clearInterval(this.heartbeatTimer);
      this.scheduleReconnect();
    };
    this.ws.onerror = () => { this.ws?.close(); };
  }

  private scheduleReconnect() {
    if (this.reconnectTimer) return;
    const delay = Math.min(1000 * 2 ** this.reconnectAttempts, 30000);
    this.reconnectAttempts++;
    this.reconnectTimer = setTimeout(() => {
      this.reconnectTimer = null; this.connect();
    }, delay);
  }

  private send(obj: any) {
    if (this.ws?.readyState === WebSocket.OPEN) this.ws.send(JSON.stringify(obj));
  }

  subscribe(channel: string, handler: Handler) {
    if (!this.handlers.has(channel)) {
      this.handlers.set(channel, new Set());
      if (this.connected) this.send({ type:'subscribe', channel });
    }
    this.handlers.get(channel)!.add(handler);
    return () => {
      const set = this.handlers.get(channel); if (!set) return;
      set.delete(handler);
      if (set.size === 0) {
        this.handlers.delete(channel);
        if (this.connected) this.send({ type:'unsubscribe', channel });
      }
    };
  }
}

let bus: WSBus | null = null;
function getBus(): WSBus {
  if (!bus) {
    const proto = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
    const url = import.meta.env.VITE_WS_URL || `${proto}//${window.location.host}/ws`;
    bus = new WSBus(url);
  }
  return bus;
}

export function useWebSocket(channel: string, handler: Handler) {
  const handlerRef = useRef(handler);
  handlerRef.current = handler;
  useEffect(() => {
    const stable = (data: any) => handlerRef.current(data);
    return getBus().subscribe(channel, stable);
  }, [channel]);
}

async function buildEarningsSection(user) {

    SELECT symbol, report_time, features, ai_analysis
    FROM earnings_cache
    WHERE report_date = $1
      AND ai_analysis->>'verdict' != 'AVOID'

    ORDER BY (ai_analysis->>'confidence')::int DESC LIMIT 6`, [today]);
